def role access_to_external
    description = 'Access to the external services'
    includes Internet
    includes admin
    includes office
    includes staff_1
    includes staff_2
    includes pool
    includes lab
    includes slb
end

def role access_to_internal
    description = 'Access to the internal services'
    includes admin
    includes office
    includes staff_1
    includes staff_2
    includes pool
    includes lab
    includes slb
end

def role access_from_admin
    description = 'Access from the admin network'
    includes internal
    includes office
    includes staff_1
    includes staff_2
    includes pool
    includes lab
    includes slb
end

def role access_to_internet
    description = 'Access to the internet'
    includes admin
    includes office
    includes staff_1
    includes staff_2
    includes pool
    includes lab
end


def policies (default: deny)
    # Access to the external services
    access_to_external  <->>  external

    # Access to the internal services
    access_to_internal  <->>  internal

    # Access from the admin network
    admin  <->>  access_from_admin

    # Access to the internet
    access_to_internet  <->>  Internet
end
